8def _format_assignment(counter_model: CounterModel) -> str: 

9 """Render a counter-model as ``P=true, Q=false`` style fragments.""" 

10 if not counter_model.assignment: 

11 return "(empty assignment — the formula is satisfied vacuously)" 

12 parts = [ 

13 f"{atom}={'true' if value else 'false'}" 

14 for atom, value in sorted(counter_model.assignment.items()) 

15 ] 

16 return ", ".join(parts)

19def format_counter_model_prompt( 

20 step: str, 

21 counter_model: CounterModel | None, 

22 premises: list[str], 

23 *, 

24 step_index: int | None = None, 

25) -> str: 

26 """Build a neutral, pedagogical correction prompt for the LLM. 

27 

28 The prompt explains which step is inconsistent, recaps the current 

29 premise list, and asks for a rewrite of just that step. When the 

30 verifier ran in entailment mode and surfaced a witness assignment, 

31 that concrete counter-model is shown to the LLM; consistency-mode 

32 rejections (UNSAT — there is no satisfying assignment) instead get a 

33 generic explanation. Tone is intentionally non-adversarial — no 

34 "you are wrong" language — so retries don't push the model toward 

35 defensive or evasive rewordings. 

36 """ 

37 label = f"step {step_index}" if step_index is not None else "this step" 

38 premise_block = "\n".join(f" - {p}" for p in premises) if premises else " (no prior premises)" 

39 

40 if counter_model is None: 

41 explanation = ( 

42 "The step contradicts the established premises (no consistent " 

43 "assignment of the underlying atoms exists)." 

44 ) 

45 else: 

46 assignment = _format_assignment(counter_model) 

47 explanation = ( 

48 f"A counter-example: assigning {assignment} satisfies the premises " 

49 f"but makes {label} false." 

50 ) 

51 

52 return ( 

53 f'The reasoning {label} "{step}" is inconsistent with the premises ' 

54 f"established so far.\n" 

55 f"\n" 

56 f"Premises so far:\n" 

57 f"{premise_block}\n" 

58 f"\n" 

59 f"{explanation} Please reconsider {label} and rewrite it as one short " 

60 f"paragraph. Do not restate the premises; emit only the corrected step." 

61 )
