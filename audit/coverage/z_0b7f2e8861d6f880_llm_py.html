13# Gemma 4 thinking-mode channel markers, derived from the actual emitted 

14# special tokens (see scripts/dump_gemma4_thinking.py). The model uses 

15# *asymmetric* brackets — opening tags are <|...> and closing tags are 

16# <...|> — and the thinking and answer phases are wrapped like this: 

17# <|channel>thought\n[Internal reasoning]<channel|>[Final answer]<turn|> 

18# Token IDs in google/gemma-4-E4B-it's tokenizer: 

19# 100 '<|channel>' - thinking channel start 

20# 101 '<channel|>' - thinking channel end / answer start 

21# 106 '<turn|>' - end-of-turn marker (must be stripped from answer) 

22THINKING_START_MARKER = "<|channel>thought\n" 

23THINKING_END_MARKER = "<channel|>" 

24ANSWER_END_MARKER = "<turn|>"

44def _split_thinking_answer( 

45 raw: str, 

46 *, 

47 start_marker: str = THINKING_START_MARKER, 

48 end_marker: str = THINKING_END_MARKER, 

49 answer_end_marker: str = ANSWER_END_MARKER, 

50) -> tuple[str, str]: 

51 """Split a complete Gemma 4 thinking-mode response into (thinking, answer). 

52 

53 Recognizes the channel-token structure 

54 ``<|channel>thought\\n...<channel|>...<turn|>``. When ``start_marker`` is 

55 absent the entire string is treated as the answer; when ``start_marker`` 

56 is present but ``end_marker`` never appears after it, every byte after 

57 the start marker is treated as thinking and the answer comes back empty. 

58 The trailing ``answer_end_marker`` (and everything after it) is stripped 

59 from the answer half. Both halves are stripped of surrounding whitespace. 

60 """ 

61 

62 def _trim_answer(text: str) -> str: 

63 if answer_end_marker and answer_end_marker in text: 

64 text = text.split(answer_end_marker, 1)[0] 

65 return text.strip() 

66 

67 if start_marker not in raw: 

68 return "", _trim_answer(raw) 

69 after_start = raw.split(start_marker, 1)[1] 

70 if end_marker not in after_start: 

71 return after_start.strip(), "" 

72 thinking, answer = after_start.split(end_marker, 1) 

73 return thinking.strip(), _trim_answer(answer)

76def _stream_with_phase( 

77 text_stream: Iterator[str], 

78 *, 

79 start_marker: str = THINKING_START_MARKER, 

80 end_marker: str = THINKING_END_MARKER, 

81 answer_end_marker: str = ANSWER_END_MARKER, 

82) -> Iterator[StreamChunk]: 

83 """Wrap a raw text-chunk stream into phase-tagged :class:`StreamChunk` events. 

84 

85 Three states: 

86 

87 1. ``pre-thinking`` — searching for ``start_marker``. Bytes seen here are 

88 discarded (Gemma 4 with ``enable_thinking=True`` emits the start 

89 marker first thing, so this state is empty in practice; non-thinking 

90 output where the start marker never arrives is treated as a final 

91 answer at end-of-stream). 

92 2. ``thinking`` — searching for ``end_marker`` and emitting buffered 

93 text as ``thinking`` phase. 

94 3. ``answer`` — emitting buffered text as ``answer`` phase, with the same 

95 lookback-buffering pattern so we can detect ``answer_end_marker`` and 

96 stop streaming as soon as it appears. 

97 

98 The lookback window per state ensures the markers are detected even when 

99 a chunk boundary falls inside one. 

100 """ 

101 state = "pre-thinking" 

102 pending = "" 

103 start_lookback = max(0, len(start_marker) - 1) 

104 end_lookback = max(0, len(end_marker) - 1) 

105 answer_end_lookback = max(0, len(answer_end_marker) - 1) if answer_end_marker else 0 

106 answer_terminated = False 

107 

108 for chunk in text_stream: 

109 if answer_terminated: 

110 break 

111 

112 pending += chunk 

113 

114 if state == "pre-thinking": 

115 if start_marker in pending: 

116 _before, after = pending.split(start_marker, 1) 

117 pending = after 

118 state = "thinking" 

119 # fall through to thinking-state handling below 

120 else: 

121 if len(pending) > start_lookback: 121 ↛ 122line 121 didn't jump to line 122 because the condition on line 121 was never true

122 pending = pending[-start_lookback:] if start_lookback else "" 

123 continue 

124 

125 if state == "thinking": 

126 if end_marker in pending: 

127 before, after = pending.split(end_marker, 1) 

128 if before: 

129 yield StreamChunk(text=before, phase="thinking") 

130 state = "answer" 

131 pending = after 

132 # fall through to answer-state handling 

133 else: 

134 if len(pending) > end_lookback: 

135 emit_now = pending[:-end_lookback] if end_lookback else pending 

136 pending = pending[-end_lookback:] if end_lookback else "" 

137 if emit_now: 137 ↛ 139line 137 didn't jump to line 139 because the condition on line 137 was always true

138 yield StreamChunk(text=emit_now, phase="thinking") 

139 continue 

140 

141 if state == "answer": 141 ↛ 108line 141 didn't jump to line 108 because the condition on line 141 was always true

142 if answer_end_marker and answer_end_marker in pending: 

143 before, _after = pending.split(answer_end_marker, 1) 

144 if before: 

145 yield StreamChunk(text=before, phase="answer") 

146 answer_terminated = True 

147 pending = "" 

148 continue 

149 

150 if len(pending) > answer_end_lookback: 

151 emit_now = pending[:-answer_end_lookback] if answer_end_lookback else pending 

152 pending = pending[-answer_end_lookback:] if answer_end_lookback else "" 

153 if emit_now: 153 ↛ 108line 153 didn't jump to line 108 because the condition on line 153 was always true

154 yield StreamChunk(text=emit_now, phase="answer") 

155 

156 if pending and not answer_terminated: 

157 if state == "pre-thinking": 157 ↛ 167line 157 didn't jump to line 167 because the condition on line 157 was always true

158 # No structured thinking ever began — treat the whole leftover 

159 # buffer as a final answer so the controller doesn't drop it. 

160 # Strip a trailing answer_end_marker if it slipped in. 

161 tail = pending 

162 if answer_end_marker and answer_end_marker in tail: 162 ↛ 163line 162 didn't jump to line 163 because the condition on line 162 was never true

163 tail = tail.split(answer_end_marker, 1)[0] 

164 if tail: 164 ↛ exitline 164 didn't return from function '_stream_with_phase' because the condition on line 164 was always true

165 yield StreamChunk(text=tail, phase="answer") 

166 else: 

167 yield StreamChunk(text=pending, phase=state) # type: ignore[arg-type]

170class StubGemmaBackend: 

171 """Deterministic backend for tests — replays a sequence of pre-recorded scenes. 

172 

173 Each call to :meth:`stream_reasoning` consumes one scene from 

174 ``scripts``. ``fail_after`` simulates mid-stream errors by raising 

175 :class:`RuntimeError` after that many chunks have been yielded. 

176 """

178 name: str = "stub.gemma"

184 fail_after: int | None = None, 

185 ) -> None: 

186 self._scripts: list[list[StreamChunk]] = list(scripts) 

187 self._fail_after: int | None = fail_after 

188 self.call_count: int = 0 

189 self.last_messages: list[dict[str, str]] = [] 

190 self.last_seed: int | None = None 

191 self.last_max_new_tokens: int | None = None 

192 

193 def stream_reasoning( 

194 self, 

195 messages: list[dict[str, str]], 

196 *, 

197 max_new_tokens: int = 4096, 

198 seed: int | None = None, 

199 ) -> Iterator[StreamChunk]: 

200 self.last_messages = list(messages) 

201 self.last_seed = seed 

202 self.last_max_new_tokens = max_new_tokens 

203 if self.call_count >= len(self._scripts): 

204 raise RuntimeError( 

205 f"StubGemmaBackend exhausted: called {self.call_count + 1} times, "

209 self.call_count += 1 

210 for i, chunk in enumerate(scene):

216class Gemma4ThinkingBackend: 

217 """Production backend: ``google/gemma-4-E4B-it`` with thinking mode enabled. 

218 

219 Both processor and model are loaded lazily on the first 

220 :meth:`stream_reasoning` call so importing this module never touches 

221 torch, transformers, or HuggingFace. Greedy decoding by default for 

222 determinism on a fixed seed; pass ``seed`` per call to seed both 

223 ``torch.manual_seed`` and ``transformers.set_seed`` before generation. 

224 

225 The ``THINKING_START_MARKER``, ``THINKING_END_MARKER``, and 

226 ``ANSWER_END_MARKER`` class attributes are the channel/turn tokens that 

227 bracket the thinking section and terminate the model's response. 

228 Override on a subclass if a future Gemma revision changes them. 

229 """ 

230 

231 name: str = "gemma-4-E4B-it" 

232 THINKING_START_MARKER: str = THINKING_START_MARKER 

233 THINKING_END_MARKER: str = THINKING_END_MARKER 

234 ANSWER_END_MARKER: str = ANSWER_END_MARKER 

235 

236 def __init__( 

237 self, 

238 model_id: str = REASONER_E4B_MODEL_ID, 

239 device: str = "auto", 

240 ) -> None: 

241 self._model_id: str = model_id 

242 self._device: str = device 

243 self._processor: Any = None 

244 self._model: Any = None 

245 self._log = get_logger("qverify.controller.gemma4") 

246 

247 def _load(self) -> None: 

248 if self._model is not None: 

249 return 

250 

251 import torch 

252 from transformers import AutoModelForCausalLM, AutoProcessor 

253 

254 self._log.info("loading reasoner model %s", self._model_id) 

255 self._processor = AutoProcessor.from_pretrained(self._model_id) 

256 self._model = AutoModelForCausalLM.from_pretrained( 

257 self._model_id, 

258 torch_dtype=torch.bfloat16, 

259 device_map=self._device, 

260 ) 

261 self._model.eval() 

262 

263 def stream_reasoning( 

264 self, 

265 messages: list[dict[str, str]], 

266 *, 

267 max_new_tokens: int = 4096, 

268 seed: int | None = None, 

269 ) -> Iterator[StreamChunk]: 

270 self._load() 

271 

272 import torch 

273 from transformers import TextIteratorStreamer, set_seed 

274 

275 if seed is not None: 

276 torch.manual_seed(seed) 

277 set_seed(seed) 

278 

279 prompt = self._processor.apply_chat_template( 

280 messages, 

281 tokenize=False, 

282 add_generation_prompt=True, 

283 enable_thinking=True, 

284 ) 

285 inputs = self._processor(text=prompt, return_tensors="pt").to(self._model.device) 

286 

287 tokenizer = getattr(self._processor, "tokenizer", self._processor) 

288 streamer = TextIteratorStreamer( 

289 tokenizer, 

290 skip_prompt=True, 

291 skip_special_tokens=False, 

292 ) 

293 

294 gen_kwargs: dict[str, Any] = { 

295 **inputs, 

296 "max_new_tokens": max_new_tokens, 

297 "streamer": streamer, 

298 "do_sample": False, 

299 } 

300 

301 thread = Thread(target=self._model.generate, kwargs=gen_kwargs) 

302 thread.start() 

303 

304 try: 

305 yield from _stream_with_phase( 

306 iter(streamer), 

307 start_marker=self.THINKING_START_MARKER, 

308 end_marker=self.THINKING_END_MARKER, 

309 answer_end_marker=self.ANSWER_END_MARKER, 

310 ) 

311 finally: 

312 thread.join()
